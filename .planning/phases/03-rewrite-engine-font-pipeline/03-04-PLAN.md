```python
@dataclass(frozen=True, slots=True)
class FontVerdict:
    """editable=False, substitution=False is a refusal, reason names it.
    editable=True, substitution=True is A-6's downgrade shape (Phase 2's editable_substitution)."""
    branch_id: str
    editable: bool
    substitution: bool
    reason: str | None = None

@dataclass(frozen=True, slots=True)
class FontProgram:
    data: bytes
    length1: int | None = None
    length2: int | None = None
    length3: int | None = None

def resolve_font(font: pikepdf.Object) -> FontVerdict: ...
def embedded_font_bytes(font_descriptor: pikepdf.Object | None) -> FontProgram | None: ...

def encoding_map(font: pikepdf.Object) -> dict[int, str]:
    """code -> glyph name, base encoding overlaid with /Differences. Codes with no
    resolvable glyph name are OMITTED, never mapped to a placeholder."""

def glyph_presence(
    font_verdict: FontVerdict, code: int | str, font_program_bytes: FontProgram | None
) -> tuple[bool, bool]:
    """Returns (editable, substitution). NEVER a refusal unless font_verdict itself
    already is one (checked first). program_bytes=None ALWAYS returns (True, True).
    An unparseable program also downgrades to (True, True), never refuses. Reuse as-is."""
```
